CREATE OR REPLACE FUNCTION match_chunks(
  query_embedding vector,
  checked_in_mood_param text,
  destination_mood_param text,
  match_count integer
)
RETURNS TABLE (
  id text,
  text text,
  checked_in_mood text,
  destination_mood text,
  distance float
)
LANGUAGE sql
AS $$
  select
    c.id,
    c.text,
    c.checked_in_mood,
    c.destination_mood,
    (c.embedding <-> query_embedding) as distance
  from public.content_chunks c
  where c.embedding is not null
    and c.checked_in_mood = checked_in_mood_param
    and c.destination_mood = destination_mood_param
  order by c.embedding <-> query_embedding
  limit match_count;
$$;

Step 3 — Create the match_chunks() function

This SQL function tells Supabase how to search your vector database for the most relevant chunks for a given emotion pair.

DROP FUNCTION IF EXISTS match_chunks(vector, text, text, integer);

CREATE OR REPLACE FUNCTION match_chunks(
  query_embedding vector,
  checked_in_mood_param text,
  destination_mood_param text,
  match_count integer
)
RETURNS TABLE (
  id uuid,
  text text,
  checked_in_mood text,
  destination_mood text,
  distance float
)
LANGUAGE sql
AS $$
  SELECT
    c.id,
    c.text,
    c.checked_in_mood,
    c.destination_mood,
    (c.embedding <-> query_embedding) AS distance
  FROM public.content_chunks c
  WHERE c.embedding IS NOT NULL
    AND c.checked_in_mood = checked_in_mood_param
    AND c.destination_mood = destination_mood_param
  ORDER BY c.embedding <-> query_embedding
  LIMIT match_count;
$$;


✅ This turns Supabase into a vector retrieval engine.
When called from your app, it returns the most relevant pieces of text for the user’s mood.

Step 4 — Write the embedChunks.ts script

Before the AI can match anything, your text chunks must be embedded (vectorized).
This script sends each chunk to OpenAI, stores the returned vector, and makes your table searchable.

import OpenAI from "openai";
import { createClient } from "@supabase/supabase-js";

const openai = new OpenAI({ apiKey: process.env.OPENAI_API_KEY });
const supabase = createClient(
  process.env.NEXT_PUBLIC_SUPABASE_URL!,
  process.env.SUPABASE_SERVICE_ROLE_KEY!
);

async function main() {
  const { data: chunks } = await supabase
    .from("content_chunks")
    .select("*")
    .is("embedding", null);

  for (const chunk of chunks || []) {
    const res = await openai.embeddings.create({
      model: "text-embedding-3-small",
      input: chunk.text
    });

    const embedding = res.data[0].embedding;

    await supabase
      .from("content_chunks")
      .update({ embedding })
      .eq("id", chunk.id);

    console.log(`✅ Embedded: ${chunk.id}`);
  }
}

main();
